Executive Summary
1.1 The Problem
The current sales module in DevCRM is described as "very bland and uninteresting." Sales teams need motivation, visibility, and tools that make their work engaging and efficient. A boring CRM leads to low adoption, poor data quality, and missed opportunities.

1.2 The Solution
Transform the sales experience by adding:

Category	Key Features	Impact
Gamification	Sales leagues, achievement badges, leaderboards, target tracking	Motivation & competition
Data Quality	Data completeness scores, validation, health dashboards	Better data = better decisions
Pipeline	Visual stages, deal aging, predictive insights	Faster deal closure
Marketing	Automated drip campaigns, SMS marketing, lead nurturing	Higher conversion rates
Task Management	Smart scheduling, automated reminders, priority center	Increased productivity
Documents	Action plans, automated workflows, deadline management	Faster transactions

import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { prisma } from '@/lib/prisma';

export async function GET(request: NextRequest) {
  const session = await getServerSession(authOptions);
  if (!session) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const searchParams = request.nextUrl.searchParams;
  const period = searchParams.get('period') || 'monthly';
  const limit = parseInt(searchParams.get('limit') || '10');

  let orderBy: any;
  let pointsField: string;

  switch (period) {
    case 'monthly':
      orderBy = { currentPoints: 'desc' };
      pointsField = 'currentPoints';
      break;
    case 'quarterly':
      orderBy = { quarterPoints: 'desc' };
      pointsField = 'quarterPoints';
      break;
    case 'yearly':
      orderBy = { totalPoints: 'desc' };
      pointsField = 'totalPoints';
      break;
    default:
      orderBy = { currentPoints: 'desc' };
      pointsField = 'currentPoints';
  }

  const salesReps = await prisma.salesRep.findMany({
    include: {
      user: {
        select: {
          name: true,
          email: true
        }
      },
      tier: true
    },
    orderBy,
    take: limit
  });

  // Calculate percentage change from previous period
  const leaderboard = salesReps.map((rep, index) => {
    // Simplified - in production, calculate actual change
    const prevPoints = rep.currentPoints * 0.8; // Placeholder
    const change = prevPoints > 0 
      ? ((rep.currentPoints - prevPoints) / prevPoints) * 100 
      : 0;

    return {
      rank: index + 1,
      id: rep.id,
      name: rep.user.name,
      points: rep[pointsField as keyof typeof rep],
      tier: rep.tier,
      change: Math.round(change),
      changeDirection: change > 10 ? 'up' : change < -10 ? 'down' : 'flat'
    };
  });

  return NextResponse.json({
    success: true,
    data: {
      period,
      leaderboard,
      total: salesReps.length
    }
  });
}

import { prisma } from '@/lib/prisma';
import { redis } from '@/lib/redis';

interface DataQualityScore {
  overall: number;
  categories: {
    contacts: number;
    properties: number;
    activities: number;
    pipeline: number;
  };
  criticalMissing: {
    field: string;
    count: number;
    percentage: number;
  }[];
  trends: {
    score: number;
    date: string;
  }[];
  recommendations: string[];
}

export class DataQualityService {
  private static instance: DataQualityService;

  static getInstance(): DataQualityService {
    if (!DataQualityService.instance) {
      DataQualityService.instance = new DataQualityService();
    }
    return DataQualityService.instance;
  }

  async getOverallScore(): Promise<DataQualityScore> {
    // Check cache
    const cacheKey = 'data-quality:overall';
    const cached = await redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const [
      contactsComplete,
      propertiesComplete,
      activitiesComplete,
      pipelineComplete,
      missingFields
    ] = await Promise.all([
      this.getContactsScore(),
      this.getPropertiesScore(),
      this.getActivitiesScore(),
      this.getPipelineScore(),
      this.getMissingFields()
    ]);

    const categories = {
      contacts: contactsComplete,
      properties: propertiesComplete,
      activities: activitiesComplete,
      pipeline: pipelineComplete
    };

    const overall = Math.round(
      (contactsComplete + propertiesComplete + activitiesComplete + pipelineComplete) / 4
    );

    const result: DataQualityScore = {
      overall,
      categories,
      criticalMissing: missingFields,
      trends: await this.getTrends(),
      recommendations: this.generateRecommendations(categories, missingFields)
    };

    // Cache for 1 hour
    await redis.setex(cacheKey, 3600, JSON.stringify(result));

    return result;
  }

  private async getContactsScore(): Promise<number> {
    const total = await prisma.customer.count();
    if (total === 0) return 0;

    const withEmail = await prisma.customer.count({
      where: { email: { not: null } }
    });
    const withPhone = await prisma.customer.count({
      where: { phone: { not: null } }
    });
    const withKYC = await prisma.customer.count({
      where: { kycStatus: { not: null } }
    });

    return Math.round(((withEmail + withPhone + withKYC) / (total * 3)) * 100);
  }

  private async getPropertiesScore(): Promise<number> {
    const total = await prisma.property.count();
    if (total === 0) return 0;

    const withType = await prisma.property.count({
      where: { unitType: { not: null } }
    });
    const withPrice = await prisma.property.count({
      where: { price: { not: null } }
    });

  private async getMissingFields(): Promise<{ field: string; count: number; percentage: number }[]> {
    const total = await prisma.customer.count();
    if (total === 0) return [];

    const missingPhone = await prisma.customer.count({
      where: { phone: null }
    });
    const missingPropertyType = await prisma.property.count({
      where: { unitType: null }
    });
    const missingBudget = await prisma.lead.count({
      where: { budget: null }
    });

    return [
      {
        field: 'Phone number',
        count: missingPhone,
        percentage: Math.round((missingPhone / total) * 100)
      },
      {
        field: 'Property type',
        count: missingPropertyType,
        percentage: Math.round((missingPropertyType / await prisma.property.count()) * 100)
      },
      {
        field: 'Budget information',
        count: missingBudget,
        percentage: Math.round((missingBudget / await prisma.lead.count()) * 100)
      }
    ];
  }

  private generateRecommendations(
    categories: any,
    missingFields: any[]
  ): string[] {
    const recommendations: string[] = [];

    if (categories.activities < 70) {
      recommendations.push('Improve activity logging - log every client interaction');
    }

    if (categories.properties < 80) {
      recommendations.push('Update property details - ensure all fields are completed');
    }

    for (const field of missingFields) {
      if (field.percentage > 15) {
        recommendations.push(`Add ${field.field} for ${field.count} records`);
      }
    }

    if (recommendations.length === 0) {
      recommendations.push('Excellent data quality! Continue maintaining high standards.');
    }

    return recommendations;
  }
